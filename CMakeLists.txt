cmake_minimum_required(VERSION 3.24)
project(robotics_lab LANGUAGES CXX)

option(ROBOTICS_LAB_CPP26 "Use experimental C++26 with Clang 20 and libc++" OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

# Shared development settings; no executable or business library yet.
add_library(robotics_options INTERFACE)
target_include_directories(robotics_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_options(robotics_options INTERFACE -Wall -Wextra -Wpedantic)
if(ROBOTICS_LAB_CPP26)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR CMAKE_CXX_COMPILER_VERSION VERSION_LESS 20)
        message(FATAL_ERROR "The C++26 learning preset requires Clang 20 or newer")
    endif()
    target_compile_options(robotics_options INTERFACE -std=c++2c -stdlib=libc++)
    target_link_options(robotics_options INTERFACE -stdlib=libc++)
else()
    target_compile_features(robotics_options INTERFACE cxx_std_20)
endif()
